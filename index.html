function escapeHtml(str) {

.replace(/</g, "&lt;")

.replace(/'/g, "&#039;");
}

// Exponential Backoff Fetch Wrapper for Gemini
async function fetchGeminiWithBackoff(payload) {
const maxRetries = 5;
let delay = 1000;
// Priority: LocalStorage Custom Key > Environment key constant
const customKey = localStorage.getItem(API_KEY_STORAGE);
const activeKey = customKey || apiKey;

const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${activeKey}`;

for (let i = 0; i < maxRetries; i++) {
try {
const response = await fetch(url, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(payload)
});
if (response.ok) {
return await response.json();
}
// Handle rate limits or other server errors (429, 5xx) with retries
const errData = await response.json().catch(() => ({}));
console.warn(`Gemini Request failed (attempt ${i + 1}/${maxRetries}):`, errData);
} catch (e) {
console.warn(`Network or system error (attempt ${i + 1}/${maxRetries}):`, e);
}
await new Promise(res => setTimeout(res, delay));
delay *= 2; // Double wait time
}
throw new Error("連線至 AI 引擎失敗。已重複嘗試 5 次，請確認網路連線、圖片解析度或至右上角確認 API Key 設定。");
}

// Robust clipboard executor using textarea fallback due to iframe restrictions
function executeClipboardCopy(text, successMessage) {
const textarea = document.createElement('textarea');
textarea.value = text;
textarea.style.position = 'fixed'; // Avoid scrolling to bottom
textarea.style.opacity = '0';
document.body.appendChild(textarea);
textarea.select();
try {
const successful = document.execCommand('copy');
if (successful) {
showToast(successMessage, 'success');
} else {
throw new Error("execCommand copy was unsuccessful");
}
} catch (err) {
console.error("Failed to copy", err);
showToast('複製失敗，瀏覽器可能不支援此指令。', 'error');
}
document.body.removeChild(textarea);
}
